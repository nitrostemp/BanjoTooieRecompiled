        "$<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>,$<CONFIG:MinSizeRel>>:/clang:-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=/src>"
        "$<$<OR:$<CONFIG:Release>,$<CONFIG:RelWithDebInfo>,$<CONFIG:MinSizeRel>>:/clang:-ffile-prefix-map=${CMAKE_BINARY_DIR}=/build>"

    OUTPUT "${_continuation_generated_cpp}" "${_continuation_generated_hpp}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare_persistent_state_probe.py"
        --source "${CMAKE_CURRENT_SOURCE_DIR}/generated/funcs_1.c"
        --output-dir "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS tools/prepare_persistent_state_probe.py tools/lift_persistent_continuations.py
        generated/funcs_1.c generated/funcs_5.c
    VERBATIM)

    tools/persistent_state_probe.cpp src/persistent_state_continuation.cpp

    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"

    ERROR_QUIET RESULT_VARIABLE _tooie_git_revision_result)

    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"

# Full generated-code compilation coverage for the continuation architecture.
# This library is not linked into TooieRecompiled and does not enable state slots.
set(_persistent_generated_all)
foreach(_source IN LISTS GENERATED_C)
    get_filename_component(_filename "${_source}" NAME)
    list(APPEND _persistent_generated_all "${CMAKE_CURRENT_BINARY_DIR}/persistent_${_filename}")
endforeach()
add_custom_command(
    OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/persistent_continuation_coverage.json"
    BYPRODUCTS ${_persistent_generated_all}
        "${CMAKE_CURRENT_BINARY_DIR}/persistent_continuation_registry.cpp"
        "${CMAKE_CURRENT_BINARY_DIR}/persistent_continuation_registry.hpp"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/lift_persistent_continuations.py"
        --source-dir "${CMAKE_CURRENT_SOURCE_DIR}/generated"
        --output-dir "${CMAKE_CURRENT_BINARY_DIR}"
    DEPENDS tools/lift_persistent_continuations.py ${GENERATED_C}
    VERBATIM)
add_library(PersistentRecompiledFuncs STATIC EXCLUDE_FROM_ALL
    ${_persistent_generated_all} "${CMAKE_CURRENT_BINARY_DIR}/persistent_continuation_coverage.json"
    "${CMAKE_CURRENT_BINARY_DIR}/persistent_continuation_registry.cpp")
target_include_directories(PersistentRecompiledFuncs PRIVATE "${RUNTIME_RECOMP_INCLUDE}"
    generated src "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(PersistentRecompiledFuncs PRIVATE
    -fno-strict-aliasing -Wno-unused-variable -Wno-implicit-function-declaration)
